  func qualificationSparseClockGap(){q.sync {
    let live=HostClockSample.read()
    // Advance the controlled receipt clock past the authored future PTS first.
    // Ending exactly at that PTS would legitimately trim a zero-duration tail.
    let advanced=max(live.uptimeAfter,lastWrittenNs+40_000_000),delta=advanced-live.uptimeAfter
    _=observeHostClock(HostClockSample(uptimeBefore:advanced,continuous:live.continuous+delta,uptimeAfter:advanced))
    _=observeHostClock(HostClockSample(uptimeBefore:advanced+1,continuous:live.continuous+delta+10_000_000_001,uptimeAfter:advanced+1))
  }}
